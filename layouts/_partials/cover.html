{{- /* Returns the cover image for a page: - album (leaf bundle): front matter "cover" (file name, e.g. "Strohrolle.jpg") or first image by name - section: front matter "cover" as path to an image in one of its albums e.g. "sanssouci/Strohrolle.jpg" (relative to the section) or "content/parks/sanssouci/Strohrolle.jpg" / "/parks/sanssouci/Strohrolle.jpg" otherwise the cover of its newest album */ -}} {{- $cover := false -}} {{- if .IsPage -}} {{- with .Params.cover -}}{{- $cover = $.Resources.GetMatch (path.Base .) -}}{{- end -}} {{- if not $cover -}}{{- $cover = index (.Resources.ByType "image") 0 -}}{{- end -}} {{- else -}} {{- with .Params.cover -}} {{- $file := path.Base . -}} {{- $dir := path.Dir . -}} {{- $album := false -}} {{- if or (hasPrefix . "content/") (hasPrefix . "/") -}} {{- $album = site.GetPage (printf "/%s" (strings.TrimPrefix "/" (strings.TrimPrefix "content/" $dir))) -}} {{- else -}} {{- $album = $.GetPage $dir -}} {{- end -}} {{- with $album -}}{{- $cover = .Resources.GetMatch $file -}}{{- end -}} {{- if not $cover -}}{{- warnf "cover %q in %s nicht gefunden" . $.File.Path -}}{{- end -}} {{- end -}} {{- if not $cover -}} {{- range .RegularPagesRecursive.ByDate.Reverse -}} {{- if not $cover -}}{{- $cover = partial "cover.html" . -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $cover -}}
